commands/mcp/ — MCP Server Management Commands

Commands for managing Model Context Protocol servers

Provides /mcp UI and claude mcp add CLI for registering, enabling, disabling, and authenticating MCP servers. Handles scope management (local/user/project), OAuth flows, XAA enterprise auth, and server approval workflows.

📚 What is this & why does it exist?

The MCP commands module is the user-facing gateway for managing external tool servers that extend Claude Code's capabilities. It lets developers register new MCP servers (via stdio, HTTP, or SSE transports), toggle them on and off, configure OAuth credentials, and set up enterprise-grade Cross-App Access (XAA) authentication. Without it, Claude Code would have no way to discover or connect to external tools.

💡 Real-world analogy

Think of MCP server management like a smart-home hub. Each MCP server is a smart device (a thermostat, a door lock, a camera). The /mcp add command is pairing a new device—you tell the hub how to talk to it (WiFi, Bluetooth, Zigbee = stdio, HTTP, SSE). The approval flow is like confirming "yes, I trust this device." Scope management (local vs project) decides whether only you see the device or everyone in the household does. XAA is like enterprise SSO for your entire smart building—one badge gets you into every room.

Example 1: Registering an MCP Server via CLI

// addCommand.ts — action handler (simplified)
async function action(
  name: string,
  commandOrUrl: string,
  extraArgs: string[],
  options: AddOptions
): Promise<void> {
  const scope = ensureConfigScope(options.scope)
  const transport = ensureTransport(options.transport)

  // Validate server name: alphanumeric, hyphens, underscores only
  if (!/^[a-zA-Z0-9_-]+$/.test(name)) {
    cliError('Server name can only contain letters, numbers, hyphens, and underscores')
    process.exit(1)
  }

  if (transport === 'http' || transport === 'sse') {
    const headers = options.header ? parseHeaders(options.header) : undefined
    const oauth: McpOAuthConfig | undefined = options.clientId
      ? { clientId: options.clientId, callbackPort: options.callbackPort, xaa: options.xaa }
      : undefined

    await addMcpConfig(name, { type: transport, url: commandOrUrl, headers, oauth }, scope)

    if (options.clientSecret) {
      const secret = await readClientSecret()
      saveMcpClientSecret(name, { type: transport, url: commandOrUrl, oauth }, secret)
    }
  } else {
    const env = options.env ? parseEnvVars(options.env) : undefined
    await addMcpConfig(name, { type: 'stdio', command: commandOrUrl, args: extraArgs, env }, scope)
  }

  logEvent('tengu_mcp_add', { type: transport, scope, source: 'command' })
  cliOk(`Added ${name} to ${describeMcpConfigFilePath(scope)}`)
}

Example 2: Toggling MCP Servers in the UI

// mcp.tsx — MCPToggle component
function MCPToggle({
  action,
  target,
  onComplete,
}: {
  action: 'enable' | 'disable'
  target: string
  onComplete: (result: string) => void
}): null {
  const clients = useAppState(s => s.mcp.clients)
  const toggleMcpServer = useMcpToggleEnabled()
  const hasRun = useRef(false)

  useEffect(() => {
    if (hasRun.current) return
    hasRun.current = true

    const enable = action === 'enable'
    const filtered = clients.filter(c => c.name !== 'ide')

    if (target === 'all') {
      const toToggle = filtered.filter(c =>
        enable ? c.type === 'disabled' : c.type !== 'disabled'
      )
      toToggle.forEach(c => toggleMcpServer(c.name, enable))
      onComplete(`${enable ? 'Enabled' : 'Disabled'} ${toToggle.length} servers`)
    } else {
      const server = filtered.find(c => c.name === target)
      if (!server) { onComplete(`Server "${target}" not found`); return }
      toggleMcpServer(server.name, enable)
      onComplete(`${enable ? 'Enabled' : 'Disabled'} ${server.name}`)
    }
  }, [])

  return null
}

Example 3: XAA IdP Setup

// xaaIdpCommand.ts — setup action (simplified)
async function setupAction(options: {
  issuer: string
  clientId: string
  clientSecret?: boolean
  callbackPort?: number
}): Promise<void> {
  // Validate issuer URL (must be https, except loopback)
  const url = new URL(options.issuer)
  const isLoopback = ['localhost', '127.0.0.1', '[::1]'].includes(url.hostname)
  if (url.protocol !== 'https:' && !isLoopback) {
    cliError('Issuer must use https:// (except loopback addresses)')
    process.exit(1)
  }

  // Save non-secret config to settings
  const settings: XaaIdpSettings = {
    issuer: options.issuer,
    clientId: options.clientId,
    callbackPort: options.callbackPort,
  }
  updateSettingsForSource('userSettings', { xaaIdp: settings })

  // Store client secret in keychain if provided
  if (options.clientSecret) {
    const secret = process.env.MCP_XAA_IDP_CLIENT_SECRET
    if (!secret) {
      cliError('Set MCP_XAA_IDP_CLIENT_SECRET env var when using --client-secret')
      process.exit(1)
    }
    saveIdpClientSecret(issuerKey(options.issuer), secret)
  }

  cliOk(`XAA IdP connection configured for ${options.issuer}`)
}
