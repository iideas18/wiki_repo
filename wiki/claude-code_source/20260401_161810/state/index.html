💡 Think of it like a Building's Foundation

The State module is the concrete foundation on which every other part of Claude Code stands. Just as a building's foundation is poured once and then every floor, wall, and roof relies on its stability, the State module is populated during bootstrap and then provides an immutable, read-only surface that every other module queries. Feature flags are like the building's blueprints — they decide what rooms exist. Session state is like the current tenant registry — it tracks who's using the building right now. And bootstrap config is the original survey data that determined where the foundation goes.

Code Examples

Reading Global State

import { getBootstrapState, getSessionState, isFeatureEnabled } from './state';

// All getters return frozen (immutable) objects
const bootstrap = getBootstrapState();
console.log(`Model: ${bootstrap.model}`);      // "claude-sonnet"
console.log(`Working dir: ${bootstrap.workingDir}`);

const session = getSessionState();
console.log(`Session ID: ${session.id}`);
console.log(`Resumable: ${session.resumable}`);

// Feature flags with type-safe keys
if (isFeatureEnabled('skills.discovery')) {
  console.log('Skill discovery is enabled');
}

// Attempting to mutate throws in strict mode
// bootstrap.model = 'other'; // ❌ TypeError: Cannot assign to read only property

Initializing State at Bootstrap

import { initState, onStateReady } from './state';
import { loadConfig } from '../config/loader';
import { resolveFlags } from './feature_flags';

// Called once during CLI startup
async function bootstrap(cliArgs: string[]) {
  const config = await loadConfig(cliArgs, process.env);
  const flags = await resolveFlags(config);

  initState({
    bootstrap: {
      apiKey: config.apiKey,
      model: config.model ?? 'claude-sonnet',
      workingDir: config.workingDir ?? process.cwd(),
    },
    session: {
      id: crypto.randomUUID(),
      resumable: config.resumable ?? true,
    },
    flags,
  });

  // State is now frozen and available globally
}

// Other modules can await state readiness
onStateReady((state) => {
  console.log(`State initialized for session ${state.session.id}`);
});
